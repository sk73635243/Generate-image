    <script>
        function getRandomColor() {
            const letters = '0123456789ABCDEF';
            let color = '#';
            for (let i = 0; i < 6; i++) {
                color += letters[Math.floor(Math.random() * 16)];
            }
            return color;
        }

        function drawRandomArt(canvas) {
            const ctx = canvas.getContext('2d');
            ctx.fillStyle = getRandomColor();
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Add some random geometric shapes
            const shapesCount = Math.floor(Math.random() * 10) + 5;
            for (let i = 0; i < shapesCount; i++) {
                ctx.fillStyle = getRandomColor();
                ctx.globalAlpha = Math.random() * 0.7 + 0.1;
                
                const shapeType = Math.floor(Math.random() * 3);
                
                if (shapeType === 0) {
                    // Circle
                    ctx.beginPath();
                    ctx.arc(
                        Math.random() * canvas.width, 
                        Math.random() * canvas.height, 
                        Math.random() * 40 + 10, 
                        0, 
                        Math.PI * 2
                    );
                    ctx.fill();
                } else if (shapeType === 1) {
                    // Rectangle
                    ctx.fillRect(
                        Math.random() * canvas.width, 
                        Math.random() * canvas.height, 
                        Math.random() * 80 + 10, 
                        Math.random() * 80 + 10
                    );
                } else if (shapeType === 2) {
                    // Triangle
                    ctx.beginPath();
                    ctx.moveTo(Math.random() * canvas.width, Math.random() * canvas.height);
                    ctx.lineTo(Math.random() * canvas.width, Math.random() * canvas.height);
                    ctx.lineTo(Math.random() * canvas.width, Math.random() * canvas.height);
                    ctx.fill();
                }
            }
        }

        function generateAll() {
            const pfpCanvas = document.getElementById('pfpCanvas');
            const bannerCanvas = document.getElementById('bannerCanvas');
            
            drawRandomArt(pfpCanvas);
            drawRandomArt(bannerCanvas);
        }

        // Initial generation on load
        generateAll();
    </script>
